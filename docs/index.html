// ---------- state ---------- const state = { connected:false, apiFormat:'anthropic', apiEndpoint:'', apiKey:'', apiModel:'', history:[] // {role:'user'|'assistant', content:'...'} }; const els = { fab:document.getElementById('fabBtn'), panel:document.getElementById('connexPanel'), close:document.getElementById('closeBtn'), body:document.getElementById('connexBody'), input:document.getElementById('connexInput'), send:document.getElementById('sendBtn'), status:document.getElementById('connexStatus'), settingsBtn:document.getElementById('settingsBtn'), overlay:document.getElementById('settingsOverlay'), cancelSettings:document.getElementById('cancelSettings'), saveSettings:document.getElementById('saveSettings'), apiFormat:document.getElementById('apiFormat'), apiEndpoint:document.getElementById('apiEndpoint'), apiKey:document.getElementById('apiKey'), apiModel:document.getElementById('apiModel'), }; const SYSTEM_PROMPT = `You are Connex, the AI assistant embedded in AstraZeneca's Access 360 provider portal. You help US healthcare providers and office staff with: medication prescribing information, enrollment forms and resources, patient affordability options, and enrolling/managing patients. Keep answers concise, clinical, and professional. When information isn't something you can verify, say so and point the provider to the relevant downloadable resource instead of guessing.`; let panelOpen = false; function togglePanel(open){ panelOpen = open; els.panel.classList.toggle('hidden', !open); if(open && els.body.children.length === 0){ greet(); } } els.fab.addEventListener('click', ()=>togglePanel(!panelOpen)); els.close.addEventListener('click', ()=>togglePanel(false)); // ---------- settings modal ---------- els.settingsBtn.addEventListener('click', ()=>{ els.apiFormat.value = state.apiFormat; els.apiEndpoint.value = state.apiEndpoint; els.apiKey.value = state.apiKey; els.apiModel.value = state.apiModel; els.overlay.classList.remove('hidden'); }); els.cancelSettings.addEventListener('click', ()=>els.overlay.classList.add('hidden')); els.saveSettings.addEventListener('click', ()=>{ state.apiFormat = els.apiFormat.value; state.apiEndpoint = els.apiEndpoint.value.trim(); state.apiKey = els.apiKey.value.trim(); state.apiModel = els.apiModel.value.trim(); state.connected = !!state.apiEndpoint; els.status.classList.toggle('connected', state.connected); els.status.innerHTML = state.connected ? ` Connected to your API (${state.apiModel || 'default model'})` : ` Not connected — running in demo mode. Click ⚙ to connect your API.`; els.overlay.classList.add('hidden'); }); // ---------- message rendering ---------- function addMsg(role, text){ const div = document.createElement('div'); div.className = 'msg ' + (role==='user'?'user':'bot'); div.textContent = text; els.body.appendChild(div); els.body.scrollTop = els.body.scrollHeight; return div; } function addQuickReplies(options){ const wrap = document.createElement('div'); wrap.className = 'quick-replies'; options.forEach(opt=>{ const b = document.createElement('button'); b.className = 'qr-btn'; b.textContent = opt; b.onclick = ()=>handleUserText(opt); wrap.appendChild(b); }); els.body.appendChild(wrap); els.body.scrollTop = els.body.scrollHeight; } function addDownload(label){ const b = document.createElement('button'); b.className = 'dl-btn'; b.innerHTML = '⬇ ' + label; els.body.appendChild(b); els.body.scrollTop = els.body.scrollHeight; } function greet(){ addMsg('bot', "Hello! Welcome to your AstraZeneca provider support experience. I'm here to help you quickly find approved medication information, forms, affordability resources, enrollment guidance, or patient support status.\n\nHow may I assist you today?"); addQuickReplies(['Medications','Patient enrollment','Forms & Resources','Affordability support']); } // ---------- demo-mode conversation tree (mirrors the mockup flow) ---------- function demoResponse(text){ const t = text.toLowerCase(); if(t.includes('medication')){ addMsg('bot','Select Medication:'); addQuickReplies(['Baxfendy','Fasenra','Calquence','Faslodex']); return; } if(t === 'calquence'){ addMsg('bot','Are you looking for:'); addQuickReplies(['Prescribing Information','Enrollment Forms & Resources','Affordability Options','Enroll & Manage Patients']); return; } if(t.includes('prescribing')){ addMsg('bot','CALQUENCE (acalabrutinib): 100 mg orally every 12 hours; swallow whole, with or without food.\n• Do not chew, crush, dissolve, or cut tablets\n• Manage toxicities via interruption, dose reduction, or discontinuation; avoid in severe hepatic impairment'); addDownload('CALQUENCE Tablets Full Prescribing Information'); return; } if(t.includes('enrollment form')){ addMsg('bot','Here is the enrollment form for CALQUENCE:'); addDownload('CALQUENCE Enrollment Form'); return; } if(t.includes('affordability')){ addMsg('bot','Here is the affordability brochure for CALQUENCE:'); addDownload('CALQUENCE Affordability Brochure'); return; } if(t.includes('enroll & manage') || t.includes('patient enrollment') || t.includes('enroll a patient')){ addMsg('bot','Please provide the patient, provider, and prescription information to start a new request for support. (In the full build, this opens an enrollment form for First Name, Last Name, DOB, contact info, insurance, and diagnosis.)'); return; } if(t.includes('forms') || t.includes('resources')){ addMsg('bot','All CALQUENCE resources:'); ['Enrollment Form','Coding Resource','Distribution Sheet','Affordability Brochure','Access and Reimbursement Guide'].forEach(addDownload); return; } addMsg('bot', "I can help with medication information, forms, affordability, enrollment guidance, or patient support status. This is demo mode — connect your AI API (⚙ icon) for live, open-ended answers."); } // ---------- send handling ---------- async function handleUserText(text){ if(!text.trim()) return; addMsg('user', text); state.history.push({role:'user', content:text}); if(state.connected){ const typing = addMsg('bot','Connex is typing…'); typing.classList.add('typing'); try{ const reply = await callCustomAPI(text); typing.remove(); addMsg('bot', reply); state.history.push({role:'assistant', content:reply}); }catch(err){ typing.remove(); addMsg('bot', 'Sorry — I could not reach your API (' + err.message + '). Check the endpoint and key in settings.'); } }else{ demoResponse(text); } } els.send.addEventListener('click', ()=>{ const v = els.input.value; els.input.value = ''; handleUserText(v); }); els.input.addEventListener('keydown', e=>{ if(e.key === 'Enter'){ const v = els.input.value; els.input.value = ''; handleUserText(v); } }); // ---------- calling the user's own API ---------- async function callCustomAPI(userText){ if(state.apiFormat === 'anthropic'){ const res = await fetch(state.apiEndpoint, { method:'POST', headers:{ 'Content-Type':'application/json', ...(state.apiKey ? {'x-api-key': state.apiKey, 'anthropic-version':'2023-06-01'} : {}) }, body: JSON.stringify({ model: state.apiModel || 'claude-sonnet-4-6', max_tokens: 1000, system: SYSTEM_PROMPT, messages: state.history }) }); if(!res.ok) throw new Error('HTTP ' + res.status); const data = await res.json(); const block = (data.content || []).find(b=>b.type==='text'); return block ? block.text : '(no text response)'; } else { const res = await fetch(state.apiEndpoint, { method:'POST', headers:{ 'Content-Type':'application/json', ...(state.apiKey ? {'Authorization':'Bearer ' + state.apiKey} : {}) }, body: JSON.stringify({ model: state.apiModel || 'gpt-4o', messages: [{role:'system', content:SYSTEM_PROMPT}, ...state.history] }) }); if(!res.ok) throw new Error('HTTP ' + res.status); const data = await res.json(); return data.choices?.[0]?.message?.content || '(no text response)'; } } // medication cards open the chat with that med preselected document.querySelectorAll('.med-card').forEach(card=>{ card.addEventListener('click', ()=>{ const name = card.querySelector('.med-name').textContent.trim(); togglePanel(true); setTimeout(()=>handleUserText(name), 150); }); });
